{
    radius = 0.00d

    _public _init_( double r )
    {
        this.radius = r;
    }
    _public override double area( )
    {
        ret 3.14d * this.radius * this.radius
    }
    _public override string name( )
    {
        ret "Circle"
    }

class Product  interface IComparable, IDescribable
{
    pid = 0i
    pname = ""
    price = 0.00d

    _public _init_( int id,  string n,  double p )
    {
        this.pid = id;
        this.pname = n;
        this.price = p;
    }
    _public override int compareTo( object other )
    {
        Product o = otheras Product;
        if o == null
        {
            ret 1i
        }
        if this.price > o.price
        {
            ret 1i
        }
        elif this.price < o.price
        {
            ret - - 1i
        }
        else
        {
            ret 0i
        }
    }
    _public override string describe( )
    {
        ret "Product[id=" + this.pid.toString() + ", name=" + this.pname + ", price=" + this.price.toString() + "]"
    }
}
