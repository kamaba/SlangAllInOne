    _public static throws Int32 sfEchoOnce( Net.TcpServer srv )
    {
        Net.TcpStream c = srv.accept();
        ByteBuffer acc = ByteBuffer(64i);
        while acc.readableBytes < 4i
        {
            ByteBuffer buf = ByteBuffer(64i);
            Int32 n = c.read(buf);
            if n <= 0i
            {
                break
            }
            acc.writeBytes(buf);
        }
        c.write(acc);
        c.close();
        ret 1i
    }
    _public static throws Int32 sfEofOnce( Net.TcpServer srv )
