PlatformSevenTest
{
    _public static check( string name,  bool cond )
    {
        if cond
        {
            SystemPrintln("[PlatformSevenTest] " + name + " : OK");
        }
        else
        {
            SystemPrintln("[PlatformSevenTest] " + name + " : FAIL");
        }
    }
    _public static observeS( string name,  string val )
    {
        SystemPrintln("[PlatformSevenTest] " + name + " : " + val);
    }
    _public static fun( )
    {
        SystemPrintln("========== PlatformSevenTest (start) ==========");
        SystemPrintln("--- device (§12.6) ---");
        cDev = Environment.current.device;
        pDev = Environment.probe.device;
        check("device kind in enum 1..5",cDev.kind >= 1i && cDev.kind <= 5i);
        check("device cpu always present",cDev.has(Environment.Platform.device.cpu) == true);
        check("device cpu count >= 1",cDev.count(Environment.Platform.device.cpu) >= 1i);
        check("device fake kind has(99) == false",cDev.has(99i) == false);
        check("device fake kind count(99) == 0",cDev.count(99i) == 0i);
        check("device kind is a present kind",cDev.has(cDev.kind) == true && cDev.count(cDev.kind) >= 1i);
        check("device probe kind is a present kind (raw)",pDev.has(pDev.kind) == true && pDev.count(pDev.kind) >= 1i);
        check("device current/probe mirror (cpu count)",pDev.count(Environment.Platform.device.cpu) == cDev.count(Environment.Platform.device.cpu));
        fakeInfo = cDev.info(99i);
        check("fake info vendor == empty",fakeInfo.vendor == "");
        check("fake info name == empty",fakeInfo.name == "");
        check("fake info memoryMB == 0",fakeInfo.memoryMB == 0i);
        fakeProbeInfo = pDev.info(99i);
        check("fake probe info vendor == empty",fakeProbeInfo.vendor == "");
        observeS("device kind (1=cpu 2=gpu 3=npu 4=dsp 5=fpga)",cDev.kind.toString());
        observeS("device probe kind (raw)",pDev.kind.toString());
        observeS("device count cpu/gpu/npu",cDev.count(Environment.Platform.device.cpu).toString() + "/" + cDev.count(Environment.Platform.device.gpu).toString() + "/" + cDev.count(Environment.Platform.device.npu).toString());
        observeS("device probe count cpu/gpu/npu (raw)",pDev.count(Environment.Platform.device.cpu).toString() + "/" + pDev.count(Environment.Platform.device.gpu).toString() + "/" + pDev.count(Environment.Platform.device.npu).toString());
        cpuInfo = cDev.info(Environment.Platform.device.cpu);
        observeS("cpu[0] vendor/name",cpuInfo.vendor + "/" + cpuInfo.name);
        gpuCount = cDev.count(Environment.Platform.device.gpu);
        if gpuCount > 0i
        {
            gpuInfo0 = cDev.info(Environment.Platform.device.gpu);
            gpuInfo1 = cDev.info(Environment.Platform.device.gpu,0i);
            observeS("gpu[0] vendor/name",gpuInfo0.vendor + "/" + gpuInfo0.name);
            observeS("gpu[0] computeCapability",gpuInfo0.computeCapability.toString());
            observeS("gpu[0] memoryMB",gpuInfo0.memoryMB.toString());
            check("gpu info(dev) == info(dev,0)",gpuInfo0.vendor == gpuInfo1.vendor && gpuInfo0.name == gpuInfo1.name);
            pGpuInfo = pDev.info(Environment.Platform.device.gpu);
            check("gpu current/probe mirror",pGpuInfo.vendor == gpuInfo0.vendor && pGpuInfo.name == gpuInfo0.name);
        }
        else
        {
            SystemPrintln("[PlatformSevenTest] gpu[0] : (no gpu on this host, skipped)");
        }
        SystemPrintln("========== PlatformSevenTest (end) ==========");
    }
}
