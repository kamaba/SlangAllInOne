    AssignBox
    {
        Int32 Int32value = 0i
        string stringname = "seed"
        Int32 Int32flag = 0b0011
        Int32[] Int32[]numbers = new(3i){1i2i3i}

        _public total( )
        {
            ret this.value + this.numbers[0i] + this.numbers[1i] + this.numbers[2i]
        }
        _public score( obj )
        {
            this.value = obj as Int32;
        }
        _public override string toString( )
        {
            ret "AssignBox(value=" + this.value.toString() + ", name=" + this.name + ", flag=" + this.flag.toString() + ", n0=" + this.numbers[0i].toString() + ", n1=" + this.numbers[1i].toString() + ", n2=" + this.numbers[2i].toString() + ")"
        }
    }
    Level<T>
    {
        T Tt = new()

        _public _init_( obj )
        {
            this.t = obj as T;
        }
        _public override string toString( )
        {

        }
    }
