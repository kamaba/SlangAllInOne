StringBuilderTest
{
    _public static check( string name,  bool cond )
    {
        if cond
        {
            global.println("[StringBuilderTest] " + name + " : OK");
        }
        else
        {
            global.println("[StringBuilderTest] " + name + " : FAIL");
        }
    }
    _public static testConstructors( )
    {
        b = StringBuilder();
        check("default empty",b.isEmpty);
        check("default capacity 512",b.capacity == 512i);
        check("default length 0",b.length == 0i);
        s = StringBuilder("abc");
        check("init(string) content",s.toString() == "abc");
        check("init(string) length",s.length == 3i);
        check("init(string) isNotEmpty",s.isNotEmpty);
        c = StringBuilder(64i);
        check("init(Int32) capacity",c.capacity == 64i);
        check("init(Int32) empty",c.isEmpty);
        f = StringBuilder.fromString("fac");
        check("fromString factory",f.toString() == "fac");
        check("fromString length",f.length == 3i);
    }
    _public static throws testAppend( )
    {
        b = StringBuilder();
        b.append("n=").append(42i).append(",").append(true);
        check("append chain string/int/bool",b.toString() == "n=42,true");
        Int64 big = 9000000000uL;
        b.clear();
        b.append(big);
        check("append Int64",b.toString() == "9000000000");
        Float64 f = 3.14d;
        b.clear();
        b.append(f);
        check("append Float64",b.toString() == "3.14");
        src = ByteBuffer.fromHex("c8");
        u = src.readU8();
        src.release();
        b.clear();
        b.append(u);
        check("append UInt8 numeric fallback",b.toString() == "200");
    }
    _public static testLinesAndFormat( )
    {
        b = StringBuilder();
        b.appendLine("first");
        b.appendLine();
        b.write("tail");
        check("appendLine/write content",b.toString() == "first

tail");
        check("appendLine length",b.length == 11i);
        f = StringBuilder();
        f.appendFormat("{0} + {1} = {2}","a","b","ab");
        check("appendFormat indexed",f.toString() == "a + b = ab");
        g = StringBuilder();
        g.appendFormat("x={} y={}",7i,"z");
        check("appendFormat autoincrement",g.toString() == "x=7 y=z");
    }
    _public static testUtf8AndBytes( )
    {
        b = StringBuilder("中文");
        check("utf8 length is bytes",b.length == 6i);
        check("utf8 toString roundtrip",b.toString() == "中文");
        arr = b.toUInt8Array();
        check("toUInt8Array length",arr.length == 6i);
        check("toUInt8Array first byte",arr[0i] == 228i);
    }
    _public static testClearAndRelease( )
    {
        b = StringBuilder("hello");
        check("isNotEmpty before clear",b.isNotEmpty);
        b.clear();
        check("clear empties",b.isEmpty && b.length == 0i);
        check("clear keeps capacity",b.capacity >= 5i);
        b.append("again");
        check("append after clear",b.toString() == "again");
        b.release();
        b.release();
        check("double release idempotent",true);
    }
    _public static fun( )
    {
        global.println("========== StringBuilderTest (start) ==========");
        testConstructors();
        testAppend();
        testLinesAndFormat();
        testUtf8AndBytes();
        testClearAndRelease();
        global.println("========== StringBuilderTest (end) ==========");
    }
}
