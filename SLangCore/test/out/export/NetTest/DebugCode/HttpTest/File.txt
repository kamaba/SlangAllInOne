    _public static Int32 sfFind( string text,  string sub )
    {
        Int32 tlen = SystemStringLength(text);
        Int32 slen = SystemStringLength(sub);
        if slen == 0i
        {
            ret 0i
        }
        if tlen < slen
        {
            ret - - 1i
        }
        Int32 last = tlen - slen;
        
        for Int32 i = 0i, i <= last, i = i + 1i;
        {
            bool matched = true;
            
            for Int32 j = 0i, j < slen, j = j + 1i;
            {
                if SystemStringCharCodeAt(text,i + j) != SystemStringCharCodeAt(sub,j)
                {
                    matched = false;
                    break
                }
            }
            if matched
            {
                ret i
            }
        }
        ret - - 1i
    }
    _public static Int32 sfHeaderInt( string text,  string name )
    {
        string pat = "
" + name + ":";
        Int32 pos = sfFind(text,pat);
        if pos < 0i
        {
            ret - - 1i
        }
        Int32 i = pos + SystemStringLength(pat);
        Int32 tlen = SystemStringLength(text);
        while i < tlen && SystemStringCharCodeAt(text,i) == 32i
        {
            i = i + 1i;
        }
        Int32 v = 0i;
        bool any = false;
        while i < tlen
        {
            Int32 ch = SystemStringCharCodeAt(text,i);
            if ch < 48i || ch > 57i
            {
                break
            }
            v = v * 10i + ch - 48i;
            any = true;
            i = i + 1i;
        }
        if any
        {
            ret v
        }
        ret - - 1i
    }
    _public static throws string sfReadAll( Net.NetStream s )
    {
        string text = "";
        Int32 headerEnd = - - 1i;
        Int32 bodyNeed = 0i;
        bool done = false;
        while done == false
        {
            ByteBuffer b = ByteBuffer(512i);
            Int32 n = s.read(b);
            if n == 0i
            {
                ret text
            }
            text = text + b.readString(n);
            if headerEnd < 0i
            {
                Int32 he = sfFind(text,"

");
                if he >= 0i
                {
                    headerEnd = he + 4i;
                    Int32 bn = sfHeaderInt(text,"Content-Length");
                    if bn < 0i
                    {
                        bn = 0i;
                    }
                    bodyNeed = bn;
                }
            }
            if headerEnd >= 0i
            {
                if SystemStringLength(text) >= headerEnd + bodyNeed
                {
                    done = true;
                }
            }
        }
        ret text
    }
