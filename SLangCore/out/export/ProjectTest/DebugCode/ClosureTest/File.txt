            Array<int> counts = [0i];
            function getCounts(  Array<int> arr )
                {
                    int totalLen = counts.length + arr.length;
                    Array<int> result = Array(totalLen);
                    int idx = 0i;
                    
                    for i = 0i, i < counts.length, i ++ 
                    {
                        result._setItem_(idx,counts._getItem_(i));
                        idx = idx + 1i;
                    }
                    
                    for j = 0i, j < arr.length, j ++ 
                    {
                        result._setItem_(idx,arr._getItem_(j));
                        idx = idx + 1i;
                    }
                    ret result
                }
            Array<int> merged = getCounts([10i,20i]);
            global.println("merged = " + merged.toString());
        }
