    class RecursiveClass
    {
        _public int factorial( int n )
        {

                ret 1i
            }
            ret n * this.factorial(n - 1i)
        }
        _public int fib( int n )
        {
            if n <= 0i
            {
                ret 0i
            }
            if n == 1i
            {
                ret 1i
            }
            ret this.fib(n - 1i) + this.fib(n - 2i)
        }
        _public bool isEven( int n )
        {
            if n == 0i
            {
                ret true
            }
            ret this.isOdd(n - 1i)
        }
        _public bool isOdd( int n )
        {

            {

            }

        }
