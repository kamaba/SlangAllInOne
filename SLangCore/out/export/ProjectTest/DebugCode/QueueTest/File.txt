    _public static testInterleavedStress( )
    {
        global.println("===== testInterleavedStress =====");
        Queue<int> q = new();
        int sum = 0i;
        
        for i = 1i, i <= 200i, i ++ 
        {
            q.enqueue(i);
            if i > 100i
            {
                sum += q.dequeue()
            }
        }
        global.println("length = " + q.length.toString());
        global.println("peek = " + q.peek.toString());
        global.println("rear = " + q.rear.toString());
        global.println("dequeue sum(1..100) = " + sum.toString());
        int remainSum = 0i;
        while q.isNotEmpty
        {
            remainSum += q.dequeue()
        }
        global.println("dequeue sum(101..200) = " + remainSum.toString());
        global.println("isEmpty = " + q.isEmpty.toString());
    }
