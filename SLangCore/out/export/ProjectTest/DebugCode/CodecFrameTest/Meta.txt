    _public static throws ByteBuffer cfEncodeMsg( object v )
    {
        CodecFrameMsg m = vas CodecFrameMsg;
        w = ProtocalBuffers.newWriter();
        w.writeInt32(1i,m.id);
        w.writeString(2i,m.name);
        ret ByteBuffer.fromBytes(w.toBytes())
    }
    _public static throws CodecFrameMsg cfDecodeMsg( ByteBuffer b )
    {
        r = ProtocalBuffers.newReader(b.toArray());
        Int32 id = 0i;
        string name = "";
        while ! ! r.atEnd
        {
            PbTag t = r.readTag();
            if t.fieldNumber == 1i
            {
                id = r.readInt32();
            }
            else
            {
                if t.fieldNumber == 2i
                {
                    name = r.readString();
                }
            }
        }
        ret CodecFrameMsg(id,name)
    }
    _public static throws ByteBuffer cfBinEncode( object v )
    {
        CodecFrameMsg m = vas CodecFrameMsg;
        b = ByteBuffer(64i);
        b.writeVarUint(SystemConvertInt64(m.id));

        ret b
    }
    _public static throws CodecFrameMsg cfBinDecode( ByteBuffer b )
    {
        u = b.readVarUint();
        Int64 id64 = SystemConvertInt64FromUInt64(u);
        Int32 id = SystemConvertInt32(id64);
        string name = b.readString(b.readableBytes);
        ret CodecFrameMsg(id,name)
    }
