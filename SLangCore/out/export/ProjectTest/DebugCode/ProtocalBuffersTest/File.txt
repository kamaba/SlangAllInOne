    _public static throws testRoundtrip( )
    {
        w = ProtocalBuffers.newWriter();
        w.writeInt32(1i,- - 42i);
        w.writeInt64(2i,10000000000uL);
        w.writeInt64(3i,- - 9876543210uL);
        w.writeSInt32(4i,- - 123456789i);
        w.writeSInt64(5i,- - 9876543210uL);
        w.writeUInt32(6i,- - 1i);
        w.writeUInt64(7i,10000000000uL);
        w.writeBool(8i,true);
        w.writeEnum(9i,3i);
        w.writeFixed32(10i,- - 1i);
        w.writeSFixed64(11i,- - 9876543210uL);
        w.writeFloat(12i,2.5f);
        w.writeDouble(13i,- - 0.5f);
        Int32 total = w.length;
        r = ProtocalBuffers.newReader(w.toBytes());
        PbTag t1 = r.readTag();
        check("rt field1 tag",t1.fieldNumber == 1i && t1.wireType == 0i);
        check("rt int32 -42",r.readInt32() == - - 42i);
        PbTag t2 = r.readTag();
        check("rt field2 tag",t2.fieldNumber == 2i);
        check("rt int64 10^10",r.readInt64() == 10000000000uL);
        PbTag t3 = r.readTag();
        check("rt int64 negative",r.readInt64() == - - 9876543210uL);
        PbTag t4 = r.readTag();
        check("rt field4 tag",t4.fieldNumber == 4i);
        check("rt sint32",r.readSInt32() == - - 123456789i);
        PbTag t5 = r.readTag();
        check("rt sint64",r.readSInt64() == - - 9876543210uL);
        PbTag t6 = r.readTag();
        check("rt field6 tag",t6.fieldNumber == 6i);
        check("rt uint32 bit pattern -1",r.readUInt32() == - - 1i);
        PbTag t7 = r.readTag();
        check("rt uint64",r.readUInt64() == 10000000000uL);
        PbTag t8 = r.readTag();
        check("rt bool true",r.readBool() == true);
        PbTag t9 = r.readTag();
        check("rt enum 3",r.readEnum() == 3i);
        PbTag t10 = r.readTag();
        check("rt field10 tag fixed32",t10.fieldNumber == 10i && t10.wireType == 5i);
        check("rt fixed32 -1",r.readFixed32() == - - 1i);
        PbTag t11 = r.readTag();
        check("rt sfixed64",r.readSFixed64() == - - 9876543210uL);
        PbTag t12 = r.readTag();
        check("rt field12 tag fixed32",t12.fieldNumber == 12i && t12.wireType == 5i);
        f = r.readFloat();
        check("rt float 2.5",f == 2.5f);
        PbTag t13 = r.readTag();
        check("rt double -0.5",r.readDouble() == - - 0.5f);
        check("rt atEnd",r.atEnd);
        check("rt position equals length",r.position == total);
    }

    _public static throws testNested( )
    {
        sub = ProtocalBuffers.newWriter();
        sub.writeInt32(1i,42i);
        sub.writeString(2i,"hi");
        check("nested sub length 6",sub.length == 6i);
        w = ProtocalBuffers.newWriter();
        w.writeInt32(1i,7i);
        w.writeMessage(2i,sub);
        check("nested outer hex",hexOf(w) == "08071206082a12026869");
        r = ProtocalBuffers.newReader(w.toBytes());
        PbTag t1 = r.readTag();
        check("nested outer field1",t1.fieldNumber == 1i && r.readInt32() == 7i);
        PbTag t2 = r.readTag();
        check("nested outer field2",t2.fieldNumber == 2i && t2.wireType == 2i);
        inner = r.readMessage();
        PbTag it1 = inner.readTag();
        check("nested inner int",it1.fieldNumber == 1i && inner.readInt32() == 42i);
        PbTag it2 = inner.readTag();
        check("nested inner string",it2.fieldNumber == 2i && inner.readString() == "hi");
        check("nested inner atEnd",inner.atEnd);
        check("nested outer atEnd",r.atEnd);
        r2 = ProtocalBuffers.newReader(w.toBytes());
        r2.readTag();
        r2.readInt32();
        r2.readTag();
        raw = r2.readMessageBytes();
        check("nested raw bytes length",raw.length == 6i);
        w2 = ProtocalBuffers.newWriter();
        w2.writeInt32(1i,7i);
        w2.writeMessage(2i,null);
        check("nested writeMessage null noop",w2.length == 2i);
    }
    _public static throws testSkip( )
    {
        w = ProtocalBuffers.newWriter();
        w.writeInt32(1i,3i);
        w.writeInt32(3i,300i);
        w.writeString(4i,"xy");
        w.writeFixed32(5i,- - 1i);
        w.writeFixed64(6i,- - 1i);
        w.writeInt32(2i,9i);
        r = ProtocalBuffers.newReader(w.toBytes());
        PbTag s1 = r.readTag();
        r.skipField(s1.wireType);
        PbTag s2 = r.readTag();
        r.skipField(s2.wireType);
        PbTag s3 = r.readTag();
        r.skipField(s3.wireType);
        PbTag s4 = r.readTag();
        r.skipField(s4.wireType);
        PbTag s5 = r.readTag();
        r.skipField(s5.wireType);
        PbTag t = r.readTag();
        check("skip reaches field2",t.fieldNumber == 2i && r.readInt32() == 9i);
        check("skip drains input",r.atEnd);
        gb = ByteBuffer.fromHex("13082a14");
        gr = ProtocalBuffers.newReader(gb.toArray());
        gb.release();
        PbTag gt = gr.readTag();
        check("group start tag",gt.fieldNumber == 2i && gt.wireType == 3i);
        gr.skipField(gt.wireType);
        check("group skip atEnd",gr.atEnd);
    }
