public _partial class MetaClass
{
    publicstring _namespaceName = ""
    publicstring _className = ""

    _public string className( )
    {
        ret this._className
    }
}
public _partial class Type extends Object
{
    publicbyte _eType = 0
    publicMetaClass _metaClass = null
    publicType[] typelist = null

    _public MetaClass metaClass( )
    {
        ret this._metaClass
    }
    _public int eType( )
    {
        ret this._eType.toInt32()
    }
    _public override string toString( )
    {
        if this._metaClass == null
        {
            ret "no_meta_class"
        }
        string str = this._metaClass.className;
        if this.typelist != null
        {
            str = str + "<";
            
            for i = 0, i < this.typelist.length, i ++ 
            {
                str = str + this.typelist[i].toString();
            }
            str = str + ">";
        }
        ret str
    }
}
