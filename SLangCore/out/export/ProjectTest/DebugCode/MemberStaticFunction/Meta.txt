class StaticMath
{
    _public static int factorial( int n )
    {
        if n <= 1i
        {
            ret 1i
        }
        ret n * StaticMath.factorial(n - 1i)
    }
    _public static int fib( int n )
    {
        if n <= 0i
        {
            ret 0i
        }
        if n == 1i
        {
            ret 1i
        }
        ret StaticMath.fib(n - 1i) + StaticMath.fib(n - 2i)
    }
    _public static int gcd( int a,  int b )
    {
        if b == 0i
        {
            ret a
        }
        ret StaticMath.gcd(b,a % b)
    }
    _public static int lcm( int a,  int b )
    {
        if a == 0i || b == 0i
        {
            ret 0i
        }
        ret a / StaticMath.gcd(a,b) * b
    }
