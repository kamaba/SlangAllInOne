    y = 0.0f

    _public float addVector( VecData2 other )

    _public float length( )

class Vector2  interface IVector2  bind VecData2
{
    VecData2 VecData2VecData2 = new()

    _public override float addVector( VecData2 other )
    {
        ret this.x + other.x + this.y + other.y
    }
    _public override float length( )
    {
        ret this.x * this.x + this.y * this.y
    }
    _public override float x( )
    {
        ret this.VecData2.x
    }
    _public override x( float _x )
    {
        this.VecData2.x = _x;
    }
    _public override float y( )
    {
        ret this.VecData2.y
    }
    _public override y( float _y )
    {
        this.VecData2.y = _y;
    }
    _public VecData2 VecData2Data( )
    {
        ret this.VecData2
    }
    _public VecData2Data( VecData2 _vecData2 )
    {
        this.VecData2 = _vecData2;
    }
    _public _init_( VecData2 _vecData2 )
    {
        this.VecData2 = _vecData2;
    }
}
