    _public static testFactory( )
    {
        b = ByteBuffer();
        check("default capacity 512",b.capacity == 512i);
        check("default maxCapacity",b.maxCapacity == 2147483647i);
        check("default indexes zero",b.readerIndex == 0i && b.writerIndex == 0i);
        b.release();
        b64 = ByteBuffer(64i);
        check("ByteBuffer(64) capacity",b64.capacity == 64i);
        b64.release();
        hb = ByteBuffer.fromHex("DeadBeef");
        check("fromHex readable",hb.readableBytes == 4i);
        check("fromHex toHex lowercase",hb.toHex() == "deadbeef");
        check("fromHex toArray length",hb.toArray().length == 4i);
        hb.release();
        sb = ByteBuffer.fromString("ABCD");
        check("fromString readable",sb.readableBytes == 4i);
        check("fromString toString",sb.toString() == "ABCD");
        sb.release();
        UInt8[] four = Array<UInt8>.create(4i);
        four[0i] = 1i;
        four[1i] = 2i;
        four[2i] = 3i;
        four[3i] = 4i;
        fb = ByteBuffer.fromBytes(four);
        check("fromBytes roundtrip",fb.toHex() == "01020304");
        wb = ByteBuffer.wrap(four);
        check("wrap roundtrip",wb.toHex() == "01020304");
        four[0i] = 99i;
        check("wrap deep copy in P0",wb.toHex() == "01020304");
    }
    _public static throws testFixedWidth( )
    {
        b = ByteBuffer(16i);
        b.writeU8(255i);
        b.writeBool(true);
        b.writeI16Le(4660i);
        b.writeI16Be(4660i);
        b.writeI32Le(305419896i);
        b.writeI32Be(305419896i);
        b.writeI64Le(1i);
        b.writeF64Be(3.1f);
        check("fixed width writerIndex 30",b.writerIndex == 30i);
        check("readU8",b.readU8() == 255i);
        check("readBool",b.readBool() == true);
        check("readI16Le",b.readI16Le() == 4660i);
        check("readI16Be",b.readI16Be() == 4660i);
        check("readI32Le",b.readI32Le() == 305419896i);
        check("readI32Be",b.readI32Be() == 305419896i);
        check("readI64Le",b.readI64Le() == 1i);
        f = b.readF64Be();
        check("readF64Be roundtrip",f == 3.1f);
        check("readerIndex drained",b.readerIndex == 30i);
        e = ByteBuffer(16i);
        e.writeI16Be(4660i);
        check("I16Be hex",e.toHex() == "1234");
        e.clear();
        e.writeI32Be(305419896i);
        check("I32Be hex",e.toHex() == "12345678");
        e.clear();
        e.writeI32Le(305419896i);
        check("I32Le hex",e.toHex() == "78563412");
        e.clear();
        e.writeU64Le(1i);
        check("U64Le hex",e.toHex() == "0100000000000000");
    }
    _public static throws testVarint( )
    {
        check("varUintEncodedSize(0)",ByteBuffer.varUintEncodedSize(0i) == 1i);
        check("varUintEncodedSize(127)",ByteBuffer.varUintEncodedSize(127i) == 1i);
        check("varUintEncodedSize(128)",ByteBuffer.varUintEncodedSize(128i) == 2i);
        check("varUintEncodedSize(16383)",ByteBuffer.varUintEncodedSize(16383i) == 2i);
        check("varUintEncodedSize(16384)",ByteBuffer.varUintEncodedSize(16384i) == 3i);
        b = ByteBuffer(16i);
        b.writeVarUint(0i);
        check("varUint(0) hex",b.toHex() == "00");
        b.clear();
        b.writeVarUint(300i);
        check("varUint(300) hex",b.toHex() == "ac02");
        b.clear();
        b.writeVarInt(- - 1i);
        check("varInt(-1) zigzag hex",b.toHex() == "01");
        b.clear();
        b.writeVarInt(123456789i);
        b.writeVarInt(- - 123456789i);
        b.writeVarUint(123456789i);
        check("varInt positive roundtrip",b.readVarInt() == 123456789i);
        check("varInt negative roundtrip",b.readVarInt() == - - 123456789i);
        check("varUint roundtrip",b.readVarUint() == 123456789i);
        t = ByteBuffer.fromHex("80");
        v = t.readVarUint();
        check("truncated varUint returns 0",v == 0i);
        check("truncated varUint no advance",t.readerIndex == 0i);
    }
    _public static throws testBulk( )
    {
        src = ByteBuffer.fromHex("0102030405");
        dst = ByteBuffer(8i);
        src.readBytes(dst,3i);
        check("readBytes dst hex",dst.toHex() == "010203");
        check("readBytes src advanced",src.readerIndex == 3i);
        arr = src.readByteArray(2i);
        check("readByteArray length",arr.length == 2i);
        check("readByteArray values",arr[0i] == 4i && arr[1i] == 5i);
        UInt8[] two = Array<UInt8>.create(2i);
        two[0i] = 4i;
        two[1i] = 5i;
        w = ByteBuffer(8i);
        w.writeByteArray(two);
        check("writeByteArray",w.toHex() == "0405");
        all = ByteBuffer(8i);
        chunk = ByteBuffer.fromHex("aabb");
        all.writeBytes(chunk);
        check("writeBytes full",all.toHex() == "aabb");
        s = ByteBuffer.fromString("helloworld");
        check("readString first half",s.readString(5i) == "hello");
        check("readString second half",s.readString(5i) == "world");
        lines = ByteBuffer.fromString("hello
world");
        l1 = lines.readLine();
        check("readLine keeps newline",l1 == "hello
");
        l2 = lines.readLine();
        check("readLine tail without newline is null",l2 == null);
    }
    _public static throws testView( )
    {
        b = ByteBuffer.fromHex("0001020304");
        s1 = b.slice(1i,3i);
        check("slice(1,3) readable",s1.toHex() == "010203");
        check("slice(1,3) capacity",s1.capacity == 3i);
        b.readerIndex = 2i;
        s2 = b.slice();
        check("slice() from readerIndex",s2.toHex() == "020304");
        d = b.duplicate();
        check("duplicate capacity",d.capacity == 5i);
        check("duplicate copies readerIndex",d.readerIndex == 2i);
        d.readerIndex = 0i;
        check("duplicate index independence",b.readerIndex == 2i);
        check("duplicate readable",d.toHex() == "0001020304");
        c = b.copy();
        check("copy capacity is readable size",c.capacity == 3i);
        check("copy readable",c.toHex() == "020304");
        c.writeU8(9i);
        check("copy independent storage",b.toHex() == "020304");
    }
    _public static throws testSearch( )
    {
        f = ByteBuffer.fromHex("01020304");
        check("indexOf byte offset from 0",f.indexOf(3i) == 2i);
        f.readerIndex = 1i;
        check("indexOf offset relative to readerIndex",f.indexOf(3i) == 1i);
        pat = ByteBuffer.fromHex("0203");
        check("indexOf pattern",f.indexOf(pat) == 0i);
        check("indexOf miss",f.indexOf(255i) == - - 1i);
        pre = ByteBuffer.fromHex("0203");
        check("startsWith true",f.startsWith(pre));
        pre2 = ByteBuffer.fromHex("0304");
        check("startsWith false",f.startsWith(pre2) == false);
        same = ByteBuffer.fromHex("020304");
        check("equals true",f.equals(same));
        diff = ByteBuffer.fromHex("0203");
        check("equals false",f.equals(diff) == false);
        check("hashCode value semantics",f.hashCode == same.hashCode);
    }
    _public static throws testCapacity( )
    {
        b = ByteBuffer(4i);
        check("writable equals capacity initially",b.writableBytes == 4i);
        b.writeU8(1i);
        b.writeU8(2i);
        check("writable after writes",b.writableBytes == 2i);
        b.ensureWritable(8i);
        check("ensureWritable grows",b.writableBytes >= 8i);
        check("ensureWritable keeps content",b.toHex() == "0102");
        g = ByteBuffer.fromHex("ff010203");
        g.readU8();
        g.discardReadBytes();
        check("discardReadBytes readerIndex",g.readerIndex == 0i);
        check("discardReadBytes content",g.toHex() == "010203");
        g.clear();
        check("clear indexes",g.readerIndex == 0i && g.writerIndex == 0i);
        check("clear readable",g.readableBytes == 0i);
        s = ByteBuffer(64i);
        s.writeU8(9i);
        s.shrink();
        check("shrink capacity",s.capacity == 1i);
        check("shrink content",s.toHex() == "09");
    }

    _public static throws testLz4( )
    {
        check("lz4 bound empty",Lz4.compressBound(0i) == 20i);
        check("lz4 bound 255",Lz4.compressBound(255i) == 276i);
        check("lz4 bound negative",Lz4.compressBound(- - 1i) == 0i);
        text = "";
        
        for Int32 i = 0i, i < 64i, i = i + 1i;
        {
            text = text + "abcdefgh";
        }
        src = ByteBuffer.fromString(text);
        packed = Lz4.compress(src);
        check("lz4 compressible shrinks",packed.readableBytes < 512i);
        back = Lz4.decompress(packed);
        check("lz4 compressible roundtrip",back.toString() == text);
        check("lz4 src index untouched",src.readerIndex == 0i);
        src.release();
        packed.release();
        back.release();
        noiseHex = "0ff1ceab2d9e37415566788a9bbccddeef0a12345678";
        noise = ByteBuffer.fromHex(noiseHex);
        npacked = Lz4.compress(noise);
        check("lz4 incompressible no shrink",npacked.readableBytes >= 26i);
        nback = Lz4.decompress(npacked);
        check("lz4 incompressible roundtrip",nback.toHex() == noiseHex);
        noise.release();
        npacked.release();
        nback.release();
        empty = ByteBuffer();
        epacked = Lz4.compress(empty);
        check("lz4 empty container hex",epacked.toHex() == "0000000000");
        check("lz4 empty container size",epacked.readableBytes == 5i);
        eback = Lz4.decompress(epacked);
        check("lz4 empty roundtrip",eback.readableBytes == 0i);
        empty.release();
        epacked.release();
        eback.release();
        shortBuf = ByteBuffer.fromHex("0102");
        caught = 0i;
        try
        {
            ;
        }
        catch
        {
            caught = 1i;
        }
        check("lz4 short header fails",caught == 1i);
        shortBuf.release();
        lieBuf = ByteBuffer.fromHex("6400000000");
        caught = 0i;
        try
        {
            ;
        }
        catch
        {
            caught = 1i;
        }
        check("lz4 lying length fails",caught == 1i);
        lieBuf.release();
        gone = ByteBuffer(4i);
        gone.writeU8(1i);
        gone.release();
        caught = 0i;
        try
        {
            ;
        }
        catch
        {
            caught = 1i;
        }
        check("lz4 released source fails",caught == 1i);
    }
