{
    _public static check( string name,  bool cond )
    {
        if cond
        {
            Console.println("[NetFrameTest] " + name + " : OK");
        }
        else
        {
            Console.println("[NetFrameTest] " + name + " : FAIL");
        }
    }
    _public static throws Int32 sfRawEcho( Net.TcpServer srv )
    {
        Net.TcpStream c = srv.accept();
        while true
        {
            ByteBuffer buf = ByteBuffer(256i);
            Int32 n = c.read(buf);
            if n <= 0i
            {
                break
            }
            c.write(buf);
        }
        c.close();
        ret 1i
    }
    _public static throws ByteBuffer sfEncFrame( object v )
    {
        string s = vas string;
        ByteBuffer payload = ByteBuffer(64i);
        payload.writeString(s);
        ByteBuffer frame = ByteBuffer(256i);
        LengthPrefix.writeFrame(frame,payload);
        ret frame
    }
    _public static throws object sfDecFrame( ByteBuffer b )
    {
        string s = b.readString(b.readableBytes);
        ret s
    }
    _public static BinaryCodec sfMakeCodec( )
    {
        var enc = function(  object v )
            {
                object r = try? try? NetFrameTest.sfEncFrame(v);
                ret r
            }
        var dec = function(  object v )
            {
                ByteBuffer b = vas ByteBuffer;
                object m = try? try? NetFrameTest.sfDecFrame(b);
                ret m
            }
        BinaryCodec k = BinaryCodec.of(enc,dec);
        ret k
    }
    _public static testGroupC( )
    {
        Console.println("========== C: LengthPrefix 分帧 ==========");
        Net.TcpServer srv1 = Net.Tcp.listen(19306i);
        var rawEchoFn1 = function(  )
            {
                object r = try? try? NetFrameTest.sfRawEcho(srv1);
            }
        Coroutine.spawnClosure0(rawEchoFn1);
        Net.TcpStream c1 = Net.Tcp.connect("127.0.0.1",19306i);
        ByteBuffer wire = ByteBuffer(256i);
        ByteBuffer p1 = ByteBuffer(64i);
        p1.writeString("hello-1");
        LengthPrefix.writeFrame(wire,p1);
        ByteBuffer p2 = ByteBuffer(64i);
        p2.writeString("hello-2");
        LengthPrefix.writeFrame(wire,p2);
        c1.write(wire);
        ByteBuffer acc1 = ByteBuffer(256i);
        Int32 frames = 0i;
        bool ok1 = false;
        bool ok2 = false;
        while frames < 2i
        {
            ByteBuffer f = LengthPrefix.tryReadFrame(acc1,1024i);
            if f != null
            {
                string s = f.readString(f.readableBytes);
                if frames == 0i
                {
                    ok1 = s == "hello-1";
                }
                else
                {
                    ok2 = s == "hello-2";
                }
                frames = frames + 1i;
            }
            else
            {
                ByteBuffer b1 = ByteBuffer(128i);
                Int32 n1 = c1.read(b1);
                if n1 <= 0i
                {
                    break
                }
                acc1.writeBytes(b1);
            }
        }
        check("C1 two frames from one write",frames == 2i && ok1 && ok2);
        c1.close();
        srv1.close();
        Net.TcpServer srv2 = Net.Tcp.listen(19307i);
        var rawEchoFn2 = function(  )
            {
                object r = try? try? NetFrameTest.sfRawEcho(srv2);
            }
        Coroutine.spawnClosure0(rawEchoFn2);
        Net.TcpStream c2 = Net.Tcp.connect("127.0.0.1",19307i);
        string msg = "half-packet-frame";
        ByteBuffer payload = ByteBuffer(64i);
        payload.writeString(msg);
        ByteBuffer frame = ByteBuffer(256i);
        LengthPrefix.writeFrame(frame,payload);
        ByteBuffer part1 = ByteBuffer(256i);
        part1.writeBytes(frame,5i);
        c2.write(part1);
        Coroutine.delay(100i);
        ByteBuffer part2 = ByteBuffer(256i);
        part2.writeBytes(frame,frame.readableBytes);
        c2.write(part2);
        ByteBuffer acc2 = ByteBuffer(256i);
        ByteBuffer got2 = null;
        while got2 == null
        {
            got2 = LengthPrefix.tryReadFrame(acc2,1024i);
            if got2 == null
            {
                ByteBuffer b2 = ByteBuffer(128i);
                Int32 n2 = c2.read(b2);
                if n2 <= 0i
                {
                    break
                }
                acc2.writeBytes(b2);
            }
        }
        bool c2ok = false;
        if got2 != null
        {
            string s2 = got2.readString(got2.readableBytes);
            c2ok = s2 == msg;
        }
        check("C2 frame split across two writes",c2ok);
        c2.close();
        srv2.close();
        Net.TcpServer srv3 = Net.Tcp.listen(19308i);
        var rawEchoFn3 = function(  )
            {
                object r = try? try? NetFrameTest.sfRawEcho(srv3);
            }
        Coroutine.spawnClosure0(rawEchoFn3);
        Net.TcpStream c3 = Net.Tcp.connect("127.0.0.1",19308i);
        BinaryCodec k = NetFrameTest.sfMakeCodec();
        Stream<object> ds = LengthPrefix.decodeStream(c3,k);
        StreamIterator<object> it = StreamIterator(ds);
        ByteBuffer f1 = NetFrameTest.sfEncFrame("alpha");
        c3.write(f1);
        ByteBuffer f2 = NetFrameTest.sfEncFrame("beta");
        c3.write(f2);
        bool m1 = it.moveNext();
        string s1 = "";
        if m1
        {
            s1 = it.currentas string;
        }
        bool m2 = it.moveNext();
        string s2 = "";
        if m2
        {
            s2 = it.currentas string;
        }
        c3.close();
        bool m3 = it.moveNext();
        check("C3 decodeStream two frames",m1 && s1 == "alpha" && m2 && s2 == "beta");
        check("C3 decodeStream ends on close",m3 == false);
        srv3.close();
    }
